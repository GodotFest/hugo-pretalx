{{/* pretalx/event-year-selector.html — Event year/prefix switcher for multi-year sites. Renders links to /{year}/talks/, /{year}/speakers/, /{year}/schedule/ for each entry in params.eventYears. Pass .context (the current page) so the current prefix and section can be detected. Site params: eventYears — slice of prefixes (e.g. ["2025", "2024"]) defaultEventYear — optional; which prefix to treat as default for "Archive" label */}} {{- $ctx := .context -}} {{- $years := site.Params.eventYears | default slice -}} {{- if eq (len $years) 0 -}} {{- $years = slice "2025" -}} {{- end -}} {{- $currentPrefix := "" -}} {{- $currentSection := "" -}} {{- if $ctx -}} {{- with $ctx.Params.pretalx_prefix -}} {{- $currentPrefix = . -}} {{- end -}} {{- if not $currentPrefix -}} {{- $pathParts := split $ctx.File.Path "/" -}} {{- if ge (len $pathParts) 2 -}} {{- $currentPrefix = index $pathParts 0 -}} {{- end -}} {{- end -}} {{- if ge (len (split $ctx.File.Path "/")) 2 -}} {{- $currentSection = index (split $ctx.File.Path "/") 1 -}} {{- end -}} {{- end -}} {{- if not $currentSection -}} {{- $currentSection = "talks" -}} {{- end -}}
Event:
{{- range $years -}} {{- $year := . -}} {{- $isCurrent := eq $year $currentPrefix -}} {{- $talksURL := printf "/%s/talks/" $year | relLangURL -}} {{- $label := $year -}} {{- if eq $year "legacy" -}} {{- $label = "Archive" -}} {{- end -}} {{- if $isCurrent -}} {{ $label }} {{- else -}} {{ $label }} {{- end -}} {{- end -}}
{{- if $currentPrefix -}} | Talks · Speakers · Schedule {{- end -}}
