{{/* pretalx/talk-card.html — Renders a single talk card for listings. Params: .talk — A talk object from the Pretalx API .prefix — The event prefix (e.g. "2025") for linking .linked — If true, wrap title in a link to the talk page (default: false) */}} {{- $talk := .talk -}} {{- $prefix := .prefix | default "" -}} {{- $linked := .linked | default false -}} {{- $title := $talk.title | default "Untitled" -}} {{- $slug := $title | urlize -}} {{- $href := "" -}} {{- if and $linked $prefix -}} {{- $href = printf "/%s/talks/%s/" $prefix $slug -}} {{- end -}}
{{/* Title */}}

{{- if $href -}} {{ $title }} {{- else -}} {{ $title }} {{- end -}}

{{/* Metadata */}}
{{- with $talk.submission_type -}} {{- $type := . -}} {{- if reflect.IsMap . -}} {{- $type = partial "pretalx/helpers.html" (dict "func" "localized" "value" . "lang" "en") -}} {{- end -}} {{ $type }} {{- end -}} {{- with $talk.duration -}} {{ . }} min {{- end -}} {{- with $talk.slot -}} {{- with .room -}} {{- $room := . -}} {{- if reflect.IsMap . -}} {{- $room = partial "pretalx/helpers.html" (dict "func" "localized" "value" . "lang" "en") -}} {{- end -}} {{ $room }} {{- end -}} {{- end -}} {{- with $talk.content_locale -}} {{ . }} {{- end -}}
{{/* Abstract excerpt */}} {{- with $talk.abstract -}}

{{ . | plainify | truncate 200 }}

{{- end -}}
{{/* Speakers */}} {{- with $talk.speakers -}}
{{- range . -}}
{{- with .avatar -}} {{- end -}} {{ .name }}
{{- end -}}
{{- end -}}
