{{/* pretalx/speaker-card.html — Renders a single speaker card. Params: .speaker — A speaker object (from Pretalx talks[].speakers or speakers endpoint) .prefix — The event prefix (e.g. "2025") for linking .linked — If true, link the speaker name to their page (default: false) .section — Optional; when set with linked, resolve href by pretalx_code so slug is correct after dedup Site param (optional): pretalxSpeakerLinksInNewTab — when true, name links use target="_blank". */}} {{- $speaker := .speaker -}} {{- $prefix := .prefix | default "" -}} {{- $linked := .linked | default false -}} {{- $section := .section -}} {{- $newTab := site.Params.pretalxSpeakerLinksInNewTab | default false -}} {{- $name := index $speaker "name" | default "Unknown Speaker" -}} {{- $code := index $speaker "code" -}} {{- $speakerPage := "" -}} {{- if and $section $code -}} {{- range $section.Pages -}} {{- if and (eq .Params.pretalx_code $code) (or (not $prefix) (eq .Params.pretalx_prefix $prefix)) -}} {{- $speakerPage = . -}} {{- break -}} {{- end -}} {{- end -}} {{- end -}} {{- $slug := replace ($name | urlize) "/" "-" -}} {{- if not $slug -}} {{- $slug = $code | lower -}} {{- end -}} {{- $href := "" -}} {{- if $linked -}} {{- $href = printf "/speakers/%s/" $slug -}} {{- with $speakerPage -}} {{- $href = .RelPermalink -}} {{- end -}} {{- end -}}
