{{/* pretalx/speaker-card.html — Renders a single speaker card. Params: .speaker — A speaker object (from Pretalx talks[].speakers or speakers endpoint) .prefix — The event prefix (e.g. "2025") for linking .linked — If true, wrap in a link to the speaker's page (default: false) .section — Optional; when set with linked, resolve href by pretalx_code so slug is correct after dedup */}} {{- $speaker := .speaker -}} {{- $prefix := .prefix | default "" -}} {{- $linked := .linked | default false -}} {{- $section := .section -}} {{- $name := index $speaker "name" | default "Unknown Speaker" -}} {{- $code := index $speaker "code" -}} {{- $href := "" -}} {{- if and $linked $prefix -}} {{- if and $section $code -}} {{- range $section.Pages -}} {{- if eq .Params.pretalx_code $code -}} {{- $href = .RelPermalink -}} {{- break -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $href -}} {{- $slug := ($name | urlize) | replace "/" "-" -}} {{- if not $slug -}} {{- $slug = $code | lower -}} {{- end -}} {{- $href = printf "/%s/speakers/%s/" $prefix $slug -}} {{- end -}} {{- end -}}

{{- end -}}
