{{/* pretalx/speaker-link.html — Link to a speaker page (e.g. in talk cards). Params: .speaker — Speaker object (code, name, avatar) .prefix — Event prefix (e.g. "2025") .section — Optional; speakers section to resolve href by pretalx_code .showAvatar — If true, show avatar before name (default: false) .plain — If true, render a instead of an . Use when the caller already wraps the whole card in a link (anchors cannot nest). Site param (optional): pretalxSpeakerLinksInNewTab — when true, uses target="_blank". */}} {{- $speaker := .speaker -}} {{- $prefix := .prefix | default "" -}} {{- $section := .section -}} {{- $showAvatar := .showAvatar | default false -}} {{- $plain := .plain | default false -}} {{- $newTab := site.Params.pretalxSpeakerLinksInNewTab | default false -}} {{- $name := index $speaker "name" | default "Unknown Speaker" -}} {{- $code := index $speaker "code" -}} {{- $speakerPage := "" -}} {{- if and $section $code -}} {{- range $section.Pages -}} {{- if and (eq .Params.pretalx_code $code) (or (not $prefix) (eq .Params.pretalx_prefix $prefix)) -}} {{- $speakerPage = . -}} {{- break -}} {{- end -}} {{- end -}} {{- end -}} {{- $slug := replace ($name | urlize) "/" "-" -}} {{- if not $slug -}} {{- $slug = $code | lower -}} {{- end -}} {{- $href := printf "/speakers/%s/" $slug -}} {{- with $speakerPage -}} {{- $href = .RelPermalink -}} {{- end -}} {{- $avatar := partial "pretalx/helpers.html" (dict "func" "image" "page" $speakerPage "spec" "144x144 webp q80" "method" "fill" "fallback" site.Params.pretalxSpeakerAvatarFallback) -}} {{- if $plain -}} {{- if and $showAvatar $avatar -}} {{- end -}} {{ $name }} {{- else -}} {{- if and $showAvatar $avatar -}} {{- end -}} {{ $name }} {{- end -}}
