"code") $code -}} {{- $speaker = . -}} {{- end -}} {{- end -}} {{- if $speaker.name -}} {{ partial "pretalx/speaker-card.html" (dict "speaker" $speaker "prefix" $prefix "linked" true) }} {{- else -}}

Speaker with code "{{ $code }}" not found in {{ $prefix }} data.

{{- end -}}
