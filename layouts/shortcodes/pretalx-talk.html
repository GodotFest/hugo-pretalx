"code") $code -}} {{- $talk = . -}} {{- end -}} {{- end -}} {{- $talksSection := site.GetPage "/talks" -}} {{- $speakersSection := site.GetPage "/speakers" -}} {{- if $talk.title -}} {{ partial "pretalx/talk-card.html" (dict "talk" $talk "prefix" $prefix "linked" true "talksSection" $talksSection "speakersSection" $speakersSection) }} {{- else -}}
